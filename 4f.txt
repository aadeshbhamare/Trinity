#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <stdlib.h>

int main()
{
    int a[100], n, i, j, temp;
    pid_t pid;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    pid = fork();

    if(pid < 0)
    {
        printf("Fork failed");
        return 1;
    }

    else if(pid == 0)
    {
        printf("\nChild Process (PID: %d)\n", getpid());

        // Bubble Sort
        for(i = 0; i < n - 1; i++)
        {
            for(j = 0; j < n - i - 1; j++)
            {
                if(a[j] > a[j+1])
                {
                    temp = a[j];
                    a[j] = a[j+1];
                    a[j+1] = temp;
                }
            }
        }

        printf("Child sorted array: ");
        for(i = 0; i < n; i++)
            printf("%d ", a[i]);

        printf("\n");
        exit(0);
    }

    else
    {
        printf("\nParent Process (PID: %d)\n", getpid());

        // Selection Sort
        for(i = 0; i < n - 1; i++)
        {
            for(j = i + 1; j < n; j++)
            {
                if(a[i] > a[j])
                {
                    temp = a[i];
                    a[i] = a[j];
                    a[j] = temp;
                }
            }
        }

        printf("Parent sorted array: ");
        for(i = 0; i < n; i++)
            printf("%d ", a[i]);

        printf("\n");

        wait(NULL);

        printf("Child process completed.\n");
    }

    return 0;
}
