
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>

int main()
{
    int fd[2];
    pid_t pid;

    char message[100];
    char buffer[100];

    // Create pipe
    if (pipe(fd) == -1)
    {
        perror("Pipe creation failed");
        return 1;
    }

    // Create child process
    pid = fork();

    if (pid < 0)
    {
        perror("Fork failed");
        return 1;
    }

    if (pid > 0)
    {
        // Parent Process

        close(fd[0]);

        printf("Enter message: ");
        fgets(message, sizeof(message), stdin);

        // Write message into pipe
        write(fd[1], message, strlen(message) + 1);

        printf("Parent: Message sent to child.\n");

        close(fd[1]);

        wait(NULL);
    }
    else
    {
        // Child Process

        close(fd[1]);

        // Read message from pipe
        read(fd[0], buffer, sizeof(buffer));

        printf("\nChild: Message received from parent: %s", buffer);

        close(fd[0]);

        exit(0);
    }

    return 0;
}
