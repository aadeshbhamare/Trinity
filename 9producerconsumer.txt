
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

#define SIZE 5

int basket[SIZE];
int in = 0, out = 0;

sem_t empty, full;
pthread_mutex_t mutex;

void *producer(void *arg)
{
    int roti = 1;

    while (roti <= 10)
    {
        sem_wait(&empty);
        pthread_mutex_lock(&mutex);

        basket[in] = roti;

        printf("Producer prepared Roti %d\n", roti);
        printf("Roti %d placed in basket\n", roti);

        in = (in + 1) % SIZE;

        pthread_mutex_unlock(&mutex);
        sem_post(&full);

        roti++;

        sleep(1);
    }

    return NULL;
}

void *consumer(void *arg)
{
    int roti;

    for (int i = 1; i <= 10; i++)
    {
        sem_wait(&full);
        pthread_mutex_lock(&mutex);

        roti = basket[out];

        printf("Consumer took Roti %d from basket\n", roti);
        printf("Consumer ate Roti %d\n\n", roti);

        out = (out + 1) % SIZE;

        pthread_mutex_unlock(&mutex);
        sem_post(&empty);

        sleep(2);
    }

    return NULL;
}

int main()
{
    pthread_t p, c;

    // Initialize semaphores and mutex
    sem_init(&empty, 0, SIZE);
    sem_init(&full, 0, 0);
    pthread_mutex_init(&mutex, NULL);

    printf("Producer-Consumer Problem (Roti Basket)\n\n");

    // Create producer and consumer threads
    pthread_create(&p, NULL, producer, NULL);
    pthread_create(&c, NULL, consumer, NULL);

    // Wait for both threads
    pthread_join(p, NULL);
    pthread_join(c, NULL);

    // Destroy semaphores and mutex
    sem_destroy(&empty);
    sem_destroy(&full);
    pthread_mutex_destroy(&mutex);

    printf("All rotis produced and consumed successfully.\n");

    return 0;
}
